CREATE TABLE salesperson (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT,
    department TEXT,
    email TEXT,
    created_at TEXT,
    updated_at TEXT
);

CREATE TABLE industry (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT,
    created_at TEXT,
    updated_at TEXT
);

CREATE TABLE customer (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT,
    industry_id INTEGER,
    address TEXT,
    phone TEXT,
    web TEXT,
    created_at TEXT,
    updated_at TEXT
);
CREATE INDEX idx_customer_industry ON customer(industry_id);

CREATE TABLE contact (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    customer_id INTEGER,
    name TEXT,
    department TEXT,
    title TEXT,
    email TEXT,
    phone TEXT,
    created_at TEXT,
    updated_at TEXT
);
CREATE INDEX idx_contact_customer ON contact(customer_id);

CREATE TABLE deal (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT,
    customer_id INTEGER,
    owner_salesperson_id INTEGER,
    status TEXT,
    probability INTEGER,
    expected_amount INTEGER,
    expected_close_date TEXT,
    close_reason TEXT,
    memo TEXT,
    created_at TEXT,
    updated_at TEXT
);
CREATE INDEX idx_deal_customer ON deal(customer_id);
CREATE INDEX idx_deal_owner ON deal(owner_salesperson_id);
CREATE INDEX idx_deal_status ON deal(status);

CREATE TABLE activity (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    deal_id INTEGER,
    type TEXT,
    activity_date_time TEXT,
    salesperson_id INTEGER,
    contact_id INTEGER,
    memo TEXT,
    created_at TEXT,
    updated_at TEXT
);
CREATE INDEX idx_activity_deal ON activity(deal_id);
CREATE INDEX idx_activity_contact ON activity(contact_id);
